function h($s) {
    return htmlspecialchars($s ?? '', ENT_QUOTES, 'UTF-8');
}

function formatSize($bytes) {
    if ($bytes <= 0) return '0 B';
    $units = ['B','KB','MB','GB','TB'];
    $i = (int)floor(log($bytes, 1024));
    return number_format($bytes / pow(1024, $i), 2) . ' ' . $units[$i];
}

function deleteItem($path) {
    if (!file_exists($path)) return false;
    if (is_dir($path)) {
        $it = new RecursiveDirectoryIterator($path, RecursiveDirectoryIterator::SKIP_DOTS);
        $files = new RecursiveIteratorIterator($it, RecursiveIteratorIterator::CHILD_FIRST);
        foreach ($files as $f) {
            is_dir($f) ? rmdir($f) : unlink($f);
        }
        return rmdir($path);
    }
    return unlink($path);
}

function createFile($dir, $fileName) {
    $path = rtrim($dir, '/') . '/' . $fileName;
    if (!file_exists($path)) {
        file_put_contents($path, "");
        header("Location: ?dir=" . urlencode($dir) . "&msg=success&text=File+created");
        exit;
    } else {
        header("Location: ?dir=" . urlencode($dir) . "&msg=error&text=File+already+exists");
        exit;
    }
}

function uploadFile($dir) {
    if (!isset($_FILES['file_to_upload']) || $_FILES['file_to_upload']['error'] != 0) {
        header("Location: ?dir=" . urlencode($dir) . "&msg=error&text=Upload+failed");
        exit;
    }
    $target = rtrim($dir, '/') . '/' . basename($_FILES['file_to_upload']['name']);
    if (move_uploaded_file($_FILES['file_to_upload']['tmp_name'], $target)) {
        header("Location: ?dir=" . urlencode($dir) . "&msg=success&text=File+uploaded");
        exit;
    } else {
        header("Location: ?dir=" . urlencode($dir) . "&msg=error&text=Upload+failed");
        exit;
    }
}

if (isset($_GET['download'])) {
    $path = realpath($_GET['download']);
    if ($path && file_exists($path) && is_file($path)) {
        header('Content-Description: File Transfer');
        header('Content-Type: application/octet-stream');
        header('Content-Disposition: attachment; filename="' . basename($path) . '"');
        header('Expires: 0');
        header('Cache-Control: must-revalidate');
        header('Pragma: public');
        header('Content-Length: ' . filesize($path));
        readfile($path);
        exit;
    }
}

if (isset($_GET['view'])) {
    $path = realpath($_GET['view']);
    if ($path && file_exists($path) && is_file($path)) {
        echo "<h2>View: " . h(basename($path)) . "</h2>";
        echo "<pre>" . h(file_get_contents($path)) . "</pre>";
        echo '<a href="?dir=' . urlencode(dirname($path)) . '" class="btn">Back</a>';
        exit;
    }
}

    $path = realpath($_GET['edit']);

        $old = realpath($_POST['rename_old']);
        $newName = trim($_POST['rename_new']);
        $new = dirname($old) . '/' . $newName;
        if ($old && file_exists($old) && !file_exists($new)) {
            rename($old, $new);
            header("Location: ?dir=" . urlencode(dirname($old)) . "&msg=success&text=Renamed");
            exit;
        } else {
            header("Location: ?dir=" . urlencode($current) . "&msg=error&text=Rename+failed");
            exit;
        }
    }
}
